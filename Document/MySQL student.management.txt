CREATE DATABASE IF NOT EXISTS student_management
  DEFAULT CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;
USE student_management;

-- =========================
-- 1) Roles (create first)
-- =========================
CREATE TABLE IF NOT EXISTS `roles` (
  `id` INT UNSIGNED NOT NULL,
  `role_name` VARCHAR(50) NOT NULL,
  `is_deleted` TINYINT(1) NOT NULL DEFAULT 0,
  `deleted_at` TIMESTAMP NULL DEFAULT NULL,
  `deleted_by` BIGINT UNSIGNED NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_role_name` (`role_name`, `id`),
  KEY `idx_roles_is_deleted` (`is_deleted`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- =========================
-- 2) Accounts (create now but add student/teacher FKs later)
--    - accounts references roles (roles exists)
--    - do NOT add fk_account_student / fk_account_teacher here to avoid circular FK
-- =========================
CREATE TABLE IF NOT EXISTS `accounts` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `username` VARCHAR(100) NOT NULL,
  `password_hash` VARCHAR(255) NOT NULL,
  `role_id` INT NOT NULL,
  `student_id` BIGINT UNSIGNED, -- FK added later
  `teacher_id` BIGINT UNSIGNED, -- FK added later
  `is_active` TINYINT(1) NOT NULL DEFAULT 1,
  `is_deleted` TINYINT(1) NOT NULL DEFAULT 0,
  `deleted_at` TIMESTAMP NULL DEFAULT NULL,
  `deleted_by` BIGINT UNSIGNED NULL,
  `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uq_username` (`username`),
  KEY `idx_accounts_is_deleted` (`is_deleted`),
  KEY `idx_account_student` (`student_id`),
  KEY `idx_account_teacher` (`teacher_id`),
  CONSTRAINT `fk_accounts_deleted_by` FOREIGN KEY (`deleted_by`) REFERENCES `accounts`(`id`) ON DELETE SET NULL,
  CONSTRAINT `fk_account_role` FOREIGN KEY (`role_id`) REFERENCES `roles`(`id`) ON DELETE RESTRICT,
  CONSTRAINT `chk_account_one_user` CHECK (
    (student_id IS NOT NULL AND teacher_id IS NULL) OR
    (student_id IS NULL AND teacher_id IS NOT NULL) OR
    (student_id IS NULL AND teacher_id IS NULL)
  )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

Những lưu ý khi sử dụng script này
Mình đã đặt deleted_by FK → accounts.id ở tất cả bảng (nếu bảng accounts bị xóa, deleted_by sẽ SET NULL).
accounts.student_id và accounts.teacher_id FK được thêm sau cùng bằng ALTER TABLE để tránh vòng tham chiếu khi tạo bảng.
Một số business rule (ví dụ: tổng assessment_weights = 1 cho mỗi class_section) không thể enforce 100% bằng SQL — xử lý trong backend (khi tạo/cập nhật weights).
password_hash phải lưu hash (bcrypt/argon2) trong accounts.password_hash.
Soft-delete workflow (backend): khi xóa mềm, cập nhật:

=============================================================================================================

UPDATE students
SET is_deleted = 1, deleted_at = NOW(), deleted_by = <account_id>
WHERE id = <student_id>;
